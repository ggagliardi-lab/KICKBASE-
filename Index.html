import requests import json # --- KONFIGURATION --- KICKBASE_EMAIL = "deine-email@example.com" KICKBASE_PASSWORD = "dein-passwort" OPENAI_API_KEY = "sk-dein-openai-api-key" # --------------------- def kickbase_login(email, password): print("🔄 Logge mich bei Kickbase ein...") url = "https://api.kickbase.com/v4/user/login" payload = { "em": email, "loy": False, "pass": password, "rep": {} } headers = {"Content-Type": "application/json"} response = requests.post(url, json=payload, headers=headers) if response.status_code != 200: raise Exception(f"Login fehlgeschlagen! Status-Code: {response.status_code}") data = response.json() token = data.get("tkn") print("✅ Erfolgreich eingeloggt!") return token def get_user_leagues(token): url = "https://api.kickbase.com/v4/leagues" headers = {"Authorization": f"Bearer {token}"} response = requests.get(url, headers=headers) return response.json().get("leagues", []) def get_league_market_and_squad(token, league_id): # Marktübersicht abrufen market_url = f"https://api.kickbase.com/v4/leagues/{league_id}/market" headers = {"Authorization": f"Bearer {token}"} market_res = requests.get(market_url, headers=headers).json() # Eidenen Kader / Team abrufen squad_url = f"https://api.kickbase.com/v4/leagues/{league_id}/me/squad" squad_res = requests.get(squad_url, headers=headers).json() return market_res, squad_res def ask_openai(prompt, data_context): print("🤖 Schicke Daten an OpenAI für die Bundesliga-Analyse...") url = "https://api.openai.com/v1/chat/completions" headers = { "Content-Type": "application/json", "Authorization": f"Bearer {OPENAI_API_KEY}" } payload = { "model": "gpt-4o-mini", "messages": [ { "role": "system", "content": "Du bist ein genialer Kickbase-Stratege und Bundesliga-Analyst. Analysiere die übertragenen Account- und Marktdaten." }, { "role": "user", "content": f"{prompt}\n\nHier sind die aktuellen Live-Daten:\n{json.dumps(data_context, indent=2)}" } ], "temperature": 0.7 } response = requests.post(url, json=payload, headers=headers) result = response.json() return result["choices"][0]["message"]["content"] if __name__ == "__main__": try: # 1. Token holen token = kickbase_login(KICKBASE_EMAIL, KICKBASE_PASSWORD) # 2. Ligen ermitteln leagues = get_user_leagues(token) if not leagues: print("❌ Keine Ligen gefunden.") exit() # Wir nehmen standardmäßig die erste Liga league_id = leagues[0]["id"] league_name = leagues[0]["name"] print(f"📌 Aktive Liga ausgewählt: {league_name} (ID: {league_id})") # 3. Markt und Kader abrufen market, squad = get_league_market_and_squad(token, league_id) # 4. KI-Analyse starten analysis = ask_openai( "Analysiere meinen Kader und den aktuellen Transfermarkt. Wer sollte aufgestellt werden? Welche Spieler auf dem Markt sind ein Must-Buy für den Gewinn der Liga?", {"market": market, "squad": squad} ) print("\n================ KI-EXPERTENRAT ================") print(analysis) print("================================================") except Exception as e: print(f"❌ Fehler: {e}")
